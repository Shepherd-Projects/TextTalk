cmake_minimum_required(VERSION 3.24)
project(TalkText VERSION 1.2.2 LANGUAGES CXX RC)
if(NOT WIN32)
  message(FATAL_ERROR "TalkText requires Windows 10 or later")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(NEMO_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/third_party/nemo-speech")
if(NOT EXISTS "${NEMO_ROOT}/include/nemo_speech/asr.h")
  message(FATAL_ERROR "Run python tools/setup.py before configuring")
endif()
add_executable(TalkText WIN32 src/main.cpp src/visuals.cpp src/gpu_visuals.cpp src/scrollbar.cpp src/platform.cpp src/speech.cpp app.rc)
target_compile_definitions(TalkText PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
target_include_directories(TalkText PRIVATE "${NEMO_ROOT}/include")
target_link_libraries(TalkText PRIVATE "${NEMO_ROOT}/lib/nemo_speech_asr_c.lib" user32 gdi32 gdiplus d3d11 d3dcompiler dxgi dcomp msimg32 shcore shell32 ole32 oleaut32 advapi32 comctl32 dwmapi uxtheme winmm uuid shlwapi)
if(MSVC)
  target_compile_options(TalkText PRIVATE /W4 /permissive- /utf-8 /EHsc)
  target_link_options(TalkText PRIVATE /MANIFEST:NO)
endif()
file(GLOB NEMO_DLLS CONFIGURE_DEPENDS "${NEMO_ROOT}/bin/*.dll")
list(FILTER NEMO_DLLS EXCLUDE REGEX "/(nemo_speech_(tts|nmt)(_c)?|llama)\\.dll$")
foreach(DLL IN LISTS NEMO_DLLS)
  add_custom_command(TARGET TalkText POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "${DLL}" "$<TARGET_FILE_DIR:TalkText>")
endforeach()
add_custom_command(TARGET TalkText POST_BUILD
  COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_CURRENT_SOURCE_DIR}/models/nemotron" "$<TARGET_FILE_DIR:TalkText>/models/nemotron")
install(TARGETS TalkText RUNTIME DESTINATION .)
install(FILES ${NEMO_DLLS} DESTINATION .)
install(DIRECTORY models/nemotron DESTINATION models)
install(FILES assets/NOTICE.txt DESTINATION licenses RENAME lucide.txt)
install(DIRECTORY "${NEMO_ROOT}/share/licenses/nemo-speech/" DESTINATION licenses/nemo-speech)
file(GLOB NEMO_NOTICES CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/third_party/licenses/nemo-*")
install(FILES ${NEMO_NOTICES} DESTINATION licenses)
option(TALKTEXT_BUILD_TESTS "Build Windows clipboard and delivery regressions" OFF)
if(TALKTEXT_BUILD_TESTS)
  add_executable(TalkTextWindowsIntegration tests/windows_integration.cpp src/platform.cpp)
  target_include_directories(TalkTextWindowsIntegration PRIVATE src)
  target_compile_definitions(TalkTextWindowsIntegration PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
  target_link_libraries(TalkTextWindowsIntegration PRIVATE user32 gdi32 shell32 ole32 oleaut32 advapi32 uuid shlwapi)
  add_executable(TalkTextClipboardStream tests/clipboard_stream.cpp)
  target_compile_definitions(TalkTextClipboardStream PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
  target_link_libraries(TalkTextClipboardStream PRIVATE user32 gdi32 gdiplus shell32 ole32 oleaut32 advapi32 uuid shlwapi)
endif()
set(CPACK_GENERATOR ZIP)
set(CPACK_PACKAGE_FILE_NAME "TalkText-${PROJECT_VERSION}-Windows-x64")
include(CPack)
